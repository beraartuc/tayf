```rust
/// Tracks which boolean axis (if any) currently has keyboard focus.
/// Parallel to `PickerSection` which only governs the color sections.
/// When `axis_focus != AxisFocus::None`, the `c` keystroke clears the
/// focused axis instead of falling through to the TrueHex hex-input
/// branch.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub(crate) enum AxisFocus {
    None,
    Bold,
    Italic,
    Underline,
}
```

```rust

//!
//! `merge_three_way` takes the disk content as it existed at the last
//! successful save (`base`), the TUI's pending-edit projection (`ours`),
//! and the current on-disk content (`theirs`). It walks all keys

/// Shape of a conflict's values. Determines how the conflict UI in
/// `widgets/conflict_list.rs` renders the row.
```

#### 3.5.2 Algoritma (rev2 — R-I6 + R-I7 fold)

Recursive walk root tables:

```text
union_keys = base_keys ∪ ours_keys ∪ theirs_keys
for k in union_keys:
    bv, ov, tv = base[k], ours[k], theirs[k]  // each Option<&Item>
    if bv == ov && bv == tv:
        keep
    elif bv == ov:
        // theirs changed → take theirs (auto-merge)
        set auto_merged[k] = tv (or remove if tv absent)
    elif bv == tv:
        // ours changed → take ours
        set auto_merged[k] = ov (or remove if ov absent)
    elif ov == tv:
        // convergent
    elif bv is Table && ov is Table && tv is Table:
        // recurse
        recurse(bv, ov, tv) with path prefix
    elif bv is ArrayOfTables && ov is ArrayOfTables && tv is ArrayOfTables:
        // R-I6: array-of-tables fallback — whole-array conflict
        push KeyConflict {
            path,
            base_value: bv.to_string(),
            ours_value: ov.to_string(),
            theirs_value: tv.to_string(),
            shape: ConflictValueShape::Block,
            is_array_block: true,
        }
        // auto_merged stays at base array
    else:
        // genuine conflict (leaf or table-shape mismatch)
        push KeyConflict {
            path,
            base_value, ours_value, theirs_value,
            shape: classify_shape(bv, ov, tv),
            is_array_block: false,
        }
        // auto_merged stays at base value
```

`classify_shape(bv, ov, tv) -> ConflictValueShape`:
- If all three are leaf-shaped (`Value::String|Integer|Float|Boolean|Datetime`) → `Leaf`.
- Otherwise (any side is Table or InlineTable or Array) → `Block`.

**Edge cases (verified against memory `feedback_toml_edit_025_quirks`):**

- **Table type drift** (one side `Table`, another `InlineTable`): treated as `Block`-shape leaf conflict (no recursion across type drift).
- **Numeric coercion** `1` vs `1.0`: `toml_edit::Item::PartialEq` distinguishes them (Integer ≠ Float). Test pin: `test_merge_numeric_one_vs_one_dot_zero_treated_as_distinct_values_documented`.
- **Comments / whitespace**: `Item` semantic equality ignores them. Comment-only theirs change → no conflict.
- **CRLF→LF on parse** (memory quirk #2): both `theirs` and `base` reparsed by toml_edit; both normalized. No conflict from line endings.
- **Nested tables**: arbitrary depth (TOML practice ≤ 3).

#### 3.6.3 `apply_conflict_selections` (R-I8 commit_save invariant share)

`save.rs` extract `commit_bytes(snapshot, body, now) -> Result<CommitOutcome, Error>` from `commit_save`. Shared 8-step ceremony: rotate backups → read disk_now → compute preserved_mode → write backup → write tmpfile → sync_all → rename → parent dir sync → reparse.

```rust
// src/config_tui/events.rs (or save.rs)
fn apply_conflict_selections(app: &mut AppState) -> Result<(), Error> {
    let SaveDiffState::MergePending { auto_merged, conflicts, selection, ours, theirs, base, .. } = &app.save_diff else { return Ok(()); };

    let mut final_doc = auto_merged.clone();
    for (conflict, choice) in conflicts.iter().zip(selection.iter()) {
        let source = match choice {
            ConflictChoice::Ours => ours,
            ConflictChoice::Theirs => theirs,
            ConflictChoice::Skip => base,
        };
        crate::config_tui::merge::write_to_path(&mut final_doc, &conflict.path, source)
            .map_err(|e| Error::Other(format!("merge apply failed: {e}")))?;
    }

    let body = final_doc.to_string();
    crate::config_tui::save::commit_bytes(&app.snapshot, &body, SystemTime::now())?;

    // Reload snapshot + clear edits (mirrors commit_save success path)
    crate::config_tui::events::request_snapshot_reload(app);
    app.edits.clear();
    app.toast = Some(Toast::ok("Saved (with manual merge)"));

    if app.pending_save_and_quit {
        app.should_quit = true;
    }
    app.save_diff = SaveDiffState::Closed;
    app.modal = Modal::None;

    Ok(())
}
```

```rust
/// - dest is a symbolic link (rename(2) would follow and overwrite outside tayf dir);
/// - dest's parent canonicalizes outside the tayf config root.
///
/// Mandate: CLAUDE.md §3 — "reject symlink traversal outside ~/.config/tayf/".
pub(crate) fn check_safe_write_destination(dest: &Path) -> Result<(), String> {
    // 1. lstat — does NOT follow symlinks
    match dest.symlink_metadata() {
        Ok(meta) if meta.file_type().is_symlink() => {
            return Err(format!("dest is a symlink: {}", dest.display()));
        }
        Ok(_) => {} // regular file or dir — dest_exists check handles separately
        Err(e) if e.kind() == std::io::ErrorKind::NotFound => {} // OK, will create
        Err(e) => return Err(format!("dest stat failed: {e}")),
    }

    // 2. Canonicalize parent (after `create_dir_all` ensures parent exists)
    let Some(parent) = dest.parent() else {
        return Err("dest has no parent".into());
    };
    std::fs::create_dir_all(parent).map_err(|e| format!("mkdir failed: {e}"))?;
    let canonical_parent = parent.canonicalize()
        .map_err(|e| format!("canonicalize parent failed: {e}"))?;
    let tayf_root = tayf_config_root()
        .ok_or("cannot resolve ~/.config/tayf/")?;
    if !canonical_parent.starts_with(&tayf_root) {
        return Err(format!(
            "parent {} is outside {}",
            canonical_parent.display(),
            tayf_root.display()
        ));
    }
    Ok(())
}

fn tayf_config_root() -> Option<PathBuf> {
    let base = std::env::var_os("XDG_CONFIG_HOME")
        .map(PathBuf::from)
        .or_else(|| std::env::var_os("HOME").map(|h| PathBuf::from(h).join(".config")))?;
    base.join("tayf").canonicalize().ok()
}
```
