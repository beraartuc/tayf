```rust
pub(crate) struct PatternsListLayout {
    pub builtin_count: usize,
    pub builtin_names: Vec<&'static str>,
    pub user_count: usize,
    pub user_names: Vec<String>,
    pub builtin_header_render_idx: usize,
    pub user_header_render_idx: usize,
}

pub(crate) fn patterns_list_layout(
    snapshot: &crate::config_tui::snapshot::ConfigSnapshot,
    filter: &str,
) -> PatternsListLayout {
    let builtin_names: Vec<&'static str> = snapshot.catalog.builtin_rule_names.iter()
        .copied()
        .filter(|n| filter.is_empty() || n.to_lowercase().contains(&filter.to_lowercase()))
        .collect();
    let builtin_set: std::collections::HashSet<&str> = snapshot.catalog.builtin_rule_names.iter().copied().collect();
    let user_names: Vec<String> = snapshot.parsed.rules.iter()
        .filter(|r| !builtin_set.contains(r.name.as_str()))
        .map(|r| r.name.clone())
        .filter(|n| filter.is_empty() || n.to_lowercase().contains(&filter.to_lowercase()))
        .collect();
    let builtin_count = builtin_names.len();
    let user_count = user_names.len();
    PatternsListLayout {
        builtin_header_render_idx: 0,
        builtin_count,
        builtin_names,
        user_header_render_idx: 1 + builtin_count,
        user_count,
        user_names,
    }
}

/// Maps `selectable_idx` (0..builtin_count + user_count) to a render-row
/// index in the rendered `Vec<ListItem>`. Skips section header rows.
pub(crate) fn selectable_to_render_idx(
    selectable_idx: usize,
    layout: &PatternsListLayout,
) -> Option<usize> {
    if selectable_idx < layout.builtin_count {
        Some(layout.builtin_header_render_idx + 1 + selectable_idx)
    } else if selectable_idx < layout.builtin_count + layout.user_count {
        let user_pos = selectable_idx - layout.builtin_count;
        Some(layout.user_header_render_idx + 1 + user_pos)
    } else {
        None
    }
}

/// Inverse mapping. Returns None when `render_idx` points at a section header.
pub(crate) fn render_to_selectable_idx(
    render_idx: usize,
    layout: &PatternsListLayout,
) -> Option<usize> {
    if render_idx == layout.builtin_header_render_idx || render_idx == layout.user_header_render_idx {
        return None;
    }
    if render_idx > layout.builtin_header_render_idx && render_idx < layout.user_header_render_idx {
        Some(render_idx - layout.builtin_header_render_idx - 1)
    } else if render_idx > layout.user_header_render_idx {
        let user_pos = render_idx - layout.user_header_render_idx - 1;
        if user_pos < layout.user_count {
            Some(layout.builtin_count + user_pos)
        } else {
            None
        }
    } else {
        None
    }
}
```

```rust
/// `$XDG_CONFIG_HOME` nor `$HOME` are set.
pub(crate) fn disk_path(name: &str) -> Option<std::path::PathBuf> {
    profiles_dir().map(|d| d.join(format!("{name}.toml")))
}

fn profiles_dir() -> Option<std::path::PathBuf> {
    let base = if let Some(xdg) = std::env::var_os("XDG_CONFIG_HOME") {
        std::path::PathBuf::from(xdg)
    } else if let Some(home) = std::env::var_os("HOME") {
        std::path::PathBuf::from(home).join(".config")
    } else {
        return None;
    };
    Some(base.join("tayf").join("profiles"))
}
```

```rust
use std::path::{Path, PathBuf};

/// Verify `dest` is safe to write through. Rejects:
/// - dest is a symbolic link (rename(2) would follow and overwrite outside tayf dir);
/// - dest's parent canonicalizes outside the tayf config root.
///
/// Mandate: CLAUDE.md §3 — "reject symlink traversal outside ~/.config/tayf/".
pub(crate) fn check_safe_write_destination(dest: &Path) -> Result<(), String> {
    // 1. lstat — does NOT follow symlinks
    match dest.symlink_metadata() {
        Ok(meta) if meta.file_type().is_symlink() => {
            return Err(format!("dest is a symlink: {}", dest.display()));
        }
        Ok(_) => {}
        Err(e) if e.kind() == std::io::ErrorKind::NotFound => {}
        Err(e) => return Err(format!("dest stat failed: {e}")),
    }

    // 2. Canonicalize parent
    let Some(parent) = dest.parent() else {
        return Err("dest has no parent".into());
    };
    std::fs::create_dir_all(parent).map_err(|e| format!("mkdir failed: {e}"))?;
    let canonical_parent = parent.canonicalize()
        .map_err(|e| format!("canonicalize parent failed: {e}"))?;
    let tayf_root = tayf_config_root()
        .ok_or_else(|| "cannot resolve ~/.config/tayf/".to_string())?;
    if !canonical_parent.starts_with(&tayf_root) {
        return Err(format!(
            "parent {} is outside {}",
            canonical_parent.display(),
            tayf_root.display()
        ));
    }
    Ok(())
}

fn tayf_config_root() -> Option<PathBuf> {
    let base = std::env::var_os("XDG_CONFIG_HOME")
        .map(PathBuf::from)
        .or_else(|| std::env::var_os("HOME").map(|h| PathBuf::from(h).join(".config")))?;
    base.join("tayf").canonicalize().ok()
}
```

### Task G7.2 — Create `merge.rs` module skeleton

```rust
//! Three-way structural merge of TOML config documents.
//!
//! `merge_three_way` takes the disk content as it existed at the last
//! successful save (`base`), the TUI's pending-edit projection (`ours`),
//! and the current on-disk content (`theirs`). It walks all keys
//! recursively and returns an auto-merged document plus the set of
//! conflicting keys.
//!
//! This module owns NO IO; it operates purely on `toml_edit::DocumentMut`.

use toml_edit::{DocumentMut, Item, Table};

pub(crate) type KeyPath = Vec<String>;

pub(crate) fn key_path_display(p: &KeyPath) -> String {
    p.join(".")
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConflictValueShape {
    Leaf,
    Block,
}

#[derive(Debug, Clone)]
pub struct KeyConflict {
    pub path: KeyPath,
    pub base_value: String,
    pub ours_value: String,
    pub theirs_value: String,
    pub shape: ConflictValueShape,
    pub is_array_block: bool,
}

#[derive(Debug, Clone)]
pub struct MergeResult {
    pub auto_merged: DocumentMut,
    pub conflicts: Vec<KeyConflict>,
}

#[derive(Debug, thiserror::Error)]
pub enum WriteToPathError {
    #[error("type mismatch at {path}: dest is {dest_type}, source is {source_type}")]
    TypeMismatch { path: String, dest_type: String, source_type: String },
    #[error("missing intermediate at {path}")]
    MissingIntermediate { path: String },
}

pub fn merge_three_way(
    base: &DocumentMut,
    ours: &DocumentMut,
    theirs: &DocumentMut,
) -> MergeResult {
    let mut auto_merged = base.clone();
    let mut conflicts = Vec::new();
    merge_table(
        base.as_table(),
        ours.as_table(),
        theirs.as_table(),
        auto_merged.as_table_mut(),
        &mut Vec::new(),
        &mut conflicts,
    );
    MergeResult { auto_merged, conflicts }
}

fn merge_table(
    base: &Table,
    ours: &Table,
    theirs: &Table,
    out: &mut Table,
    path: &mut KeyPath,
    conflicts: &mut Vec<KeyConflict>,
) {
    // Collect union of keys
    let mut keys: std::collections::BTreeSet<&str> = std::collections::BTreeSet::new();
    for k in base.iter().map(|(k, _)| k) { keys.insert(k); }
    for k in ours.iter().map(|(k, _)| k) { keys.insert(k); }
    for k in theirs.iter().map(|(k, _)| k) { keys.insert(k); }

    for k in keys {
        path.push(k.to_string());

        let bv = base.get(k);
        let ov = ours.get(k);
        let tv = theirs.get(k);

        let bv_eq_ov = items_eq(bv, ov);
        let bv_eq_tv = items_eq(bv, tv);
        let ov_eq_tv = items_eq(ov, tv);

        if bv_eq_ov && bv_eq_tv {
            // keep — no change
        } else if bv_eq_ov {
            // theirs changed → take theirs
            set_or_remove(out, k, tv);
        } else if bv_eq_tv {
            // ours changed → take ours
            set_or_remove(out, k, ov);
        } else if ov_eq_tv {
            // convergent — both changed identically
            set_or_remove(out, k, ov);
        } else if is_table(bv) && is_table(ov) && is_table(tv) {
            // recurse
            let bt = bv.and_then(|i| i.as_table()).expect("is_table");
            let ot = ov.and_then(|i| i.as_table()).expect("is_table");
            let tt = tv.and_then(|i| i.as_table()).expect("is_table");
            let entry = out.entry(k).or_insert_with(|| Item::Table(Table::new()));
            if let Some(out_table) = entry.as_table_mut() {
                merge_table(bt, ot, tt, out_table, path, conflicts);
            }
        } else if is_array_of_tables(bv) && is_array_of_tables(ov) && is_array_of_tables(tv) {
            // Array-of-tables fallback — whole-array conflict (v0.6.2 limitation)
            conflicts.push(KeyConflict {
                path: path.clone(),
                base_value: render_item(bv),
                ours_value: render_item(ov),
                theirs_value: render_item(tv),
                shape: ConflictValueShape::Block,
                is_array_block: true,
            });
        } else {
            // Genuine leaf or shape-mismatch conflict
            conflicts.push(KeyConflict {
                path: path.clone(),
                base_value: render_item(bv),
                ours_value: render_item(ov),
                theirs_value: render_item(tv),
                shape: classify_shape(bv, ov, tv),
                is_array_block: false,
            });
        }

        path.pop();
    }
}

fn items_eq(a: Option<&Item>, b: Option<&Item>) -> bool {
    match (a, b) {
        (None, None) => true,
        (Some(x), Some(y)) => format!("{x}") == format!("{y}"),
        _ => false,
    }
}

fn is_table(it: Option<&Item>) -> bool {
    matches!(it, Some(Item::Table(_)))
}

fn is_array_of_tables(it: Option<&Item>) -> bool {
    matches!(it, Some(Item::ArrayOfTables(_)))
}

fn classify_shape(b: Option<&Item>, o: Option<&Item>, t: Option<&Item>) -> ConflictValueShape {
    let all_leaf = [b, o, t].iter().all(|it| match it {
        Some(Item::Value(v)) => matches!(
            v,
            toml_edit::Value::String(_)
                | toml_edit::Value::Integer(_)
                | toml_edit::Value::Float(_)
                | toml_edit::Value::Boolean(_)
                | toml_edit::Value::Datetime(_)
        ),
        None => true,
        _ => false,
    });
    if all_leaf { ConflictValueShape::Leaf } else { ConflictValueShape::Block }
}

fn render_item(it: Option<&Item>) -> String {
    match it {
        Some(i) => format!("{i}"),
        None => "(absent)".to_string(),
    }
}

fn set_or_remove(out: &mut Table, key: &str, item: Option<&Item>) {
    match item {
        Some(i) => { out.insert(key, i.clone()); }
        None => { out.remove(key); }
    }
}

pub fn write_to_path(
    doc: &mut DocumentMut,
    path: &[String],
    source: &DocumentMut,
) -> Result<(), WriteToPathError> {
    if path.is_empty() {
        return Err(WriteToPathError::MissingIntermediate { path: "(empty)".to_string() });
    }
    let display_path = path.join(".");

    // Resolve source leaf
    let mut src_item: &Item = source.as_item();
    for seg in path {
        let Some(t) = src_item.as_table() else {
            return Err(WriteToPathError::MissingIntermediate { path: display_path.clone() });
        };
        let Some(next) = t.get(seg) else {
            return Err(WriteToPathError::MissingIntermediate { path: display_path.clone() });
        };
        src_item = next;
    }
    let src_leaf = src_item.clone();

    // Walk dest, creating intermediates as needed
    let mut cur: &mut Item = doc.as_item_mut();
    for (i, seg) in path.iter().enumerate() {
        let is_leaf = i == path.len() - 1;
        let Some(t) = cur.as_table_mut() else {
            return Err(WriteToPathError::TypeMismatch {
                path: display_path.clone(),
                dest_type: cur.type_name().to_string(),
                source_type: "table".to_string(),
            });
        };
        if is_leaf {
            t.insert(seg, src_leaf.clone());
            return Ok(());
        }
        let entry = t.entry(seg).or_insert_with(|| Item::Table(Table::new()));
        cur = entry;
    }
    Ok(())
}
```

```rust
    // Extract conflicts + selection + focused_row from SaveDiff state
    let SaveDiffState::MergePending { conflicts, selection, focused_row, .. } = &mut app.save_diff else {
        return;
    };

    // Defensive guard (N7): never % 0 panic
    if conflicts.is_empty() {
        return;
    }

    let len = conflicts.len();
    let block_at = |i: usize| matches!(conflicts[i].shape, crate::config_tui::merge::ConflictValueShape::Block);

    match (k.code, k.modifiers) {
        (KeyCode::Char('j') | KeyCode::Down, _) => *focused_row = (*focused_row + 1) % len,
        (KeyCode::Char('k') | KeyCode::Up, _) => *focused_row = (*focused_row + len - 1) % len,
        (KeyCode::Char('o'), _) => {
            if block_at(*focused_row) {
                app.toast = Some(Toast::warn("table-shaped conflict — must be resolved manually; press 's' to skip".to_string()));
                return;

            selection[*focused_row] = ConflictChoice::Ours;
```

```rust
fn apply_conflict_selections(app: &mut AppState) -> Result<(), Error> {
    let SaveDiffState::MergePending {
        auto_merged, conflicts, selection, ours, theirs, base, ..
    } = &app.save_diff else {
        return Ok(());
    };

    let mut final_doc = auto_merged.clone();
    for (i, choice) in selection.iter().enumerate() {
        let source = match choice {
            ConflictChoice::Ours => ours,
            ConflictChoice::Theirs => theirs,
            ConflictChoice::Skip => base,
        };
        crate::config_tui::merge::write_to_path(&mut final_doc, &conflicts[i].path, source)
            .map_err(|e| Error::Other(format!("write_to_path: {e}")))?;
    }

    let body = final_doc.to_string();
    let outcome = crate::config_tui::save::commit_bytes(&app.snapshot, &body, std::time::SystemTime::now())?;

    crate::config_tui::events::request_snapshot_reload(app);
    app.edits.clear();
    app.toast = Some(Toast::ok("Saved (with manual merge)".to_string()));

    if app.pending_save_and_quit {
        app.should_quit = true;
    }
    app.save_diff = SaveDiffState::Closed;
    app.modal = Modal::None;

    Ok(())
}
```
